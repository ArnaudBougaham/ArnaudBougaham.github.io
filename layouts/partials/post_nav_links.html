{{- /* Custom post navigation - only show prev/next for actual content pages in menu order */ -}} {{- $pageOrder := slice "experience" "education" "publications" "awards" "skills" "certifications" -}} {{- $pages := slice -}} {{- range $pageOrder -}} {{- $page := site.GetPage (printf "/%s" .) -}} {{- if and $page (ne $page.Kind "404") -}} {{- $pages = $pages | append $page -}} {{- end -}} {{- end -}} {{- if and (gt (len $pages) 1) -}} {{- $currentIndex := -1 -}} {{- $currentPageName := .File.LogicalName | replaceRE "\\.md$" "" -}} {{- range $i, $pageName := $pageOrder -}} {{- if eq $currentPageName $pageName -}} {{- $currentIndex = $i -}} {{- end -}} {{- end -}} {{- if ge $currentIndex 0 -}}
{{- if gt $currentIndex 0 -}} {{- $prevPage := index $pages (sub $currentIndex 1) -}} « {{ i18n "prev_page" }}
{{- $prevPage.Title -}} {{- end -}} {{- if lt $currentIndex (sub (len $pages) 1) -}} {{- $nextPage := index $pages (add $currentIndex 1) -}} {{ i18n "next_page" }} »
{{- $nextPage.Title -}} {{- end -}}
{{- end -}} {{- end }}
